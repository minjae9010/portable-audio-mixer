cmake_minimum_required(VERSION 3.24)
project(PortableMixer VERSION 0.5.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
add_executable(mixer_core_tests tests/core_tests.cpp)
target_include_directories(mixer_core_tests PRIVATE src)
add_test(NAME mixer_core_tests COMMAND mixer_core_tests)
if(MSVC)
  target_compile_options(mixer_core_tests PRIVATE /W4 /WX /permissive- /utf-8)
endif()
if(WIN32)
  add_executable(mix_monitor_tests tests/mix_monitor_tests.cpp src/mix_monitor.cpp)
  target_include_directories(mix_monitor_tests PRIVATE src)
  target_compile_definitions(mix_monitor_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(mix_monitor_tests PRIVATE ole32 uuid mmdevapi)
  add_test(NAME mix_monitor_tests COMMAND mix_monitor_tests)
  set_tests_properties(mix_monitor_tests PROPERTIES TIMEOUT 15)
  if(MSVC)
    target_compile_options(mix_monitor_tests PRIVATE /W4 /utf-8 /EHsc)
  elseif(MINGW)
    target_link_options(mix_monitor_tests PRIVATE -static)
  endif()
  add_executable(added_programs_tests tests/added_programs_tests.cpp)
  target_include_directories(added_programs_tests PRIVATE src)
  target_compile_definitions(added_programs_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(added_programs_tests PRIVATE ole32 uuid)
  add_test(NAME added_programs_tests COMMAND added_programs_tests)
  if(MSVC)
    target_compile_options(added_programs_tests PRIVATE /W4 /utf-8 /EHsc)
  elseif(MINGW)
    target_link_options(added_programs_tests PRIVATE -static)
  endif()
  if(NOT MSVC AND NOT MINGW)
    message(FATAL_ERROR "Use MSVC or LLVM-MinGW")
  endif()
  add_executable(virtual_device_tests tests/virtual_device_tests.cpp)
  target_include_directories(virtual_device_tests PRIVATE src)
  target_compile_definitions(virtual_device_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(virtual_device_tests PRIVATE ole32 uuid mmdevapi cfgmgr32 shell32 user32)
  add_test(NAME virtual_device_tests COMMAND virtual_device_tests)
  set_tests_properties(virtual_device_tests PROPERTIES TIMEOUT 15)
  if(MSVC)
    target_compile_options(virtual_device_tests PRIVATE /W4 /utf-8 /EHsc)
  endif()
  if(MINGW)
    target_link_options(virtual_device_tests PRIVATE -static)
  endif()
  enable_language(RC)
  add_executable(PortableMixer WIN32 src/main.cpp src/audio_engine.cpp src/mix_monitor.cpp src/brand.rc)
  add_executable(process_capture_tests tests/process_capture_tests.cpp src/mix_monitor.cpp)
  target_include_directories(process_capture_tests PRIVATE src)
  target_compile_definitions(process_capture_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A00000B)
  target_link_libraries(process_capture_tests PRIVATE ole32 uuid mmdevapi avrt cfgmgr32 shell32 user32)
  if(MSVC)
    target_compile_options(process_capture_tests PRIVATE /W4 /utf-8 /EHsc)
  elseif(MINGW)
    target_link_options(process_capture_tests PRIVATE -static)
  endif()
  target_include_directories(PortableMixer PRIVATE src)
  set_source_files_properties(src/brand.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/portable-mixer.ico;${CMAKE_CURRENT_SOURCE_DIR}/src/resource.h")
  include(cmake/EmbedCable.cmake)
  add_executable(bundle_tests tests/bundle_tests.cpp)
  target_include_directories(bundle_tests PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}")
  target_compile_definitions(bundle_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(bundle_tests PRIVATE bcrypt ole32 wintrust advapi32 user32 cfgmgr32 uuid mmdevapi shell32)
  add_dependencies(bundle_tests PortableMixer)
  add_test(NAME bundle_tests COMMAND bundle_tests "$<TARGET_FILE:PortableMixer>")
  if(MSVC)
    target_compile_options(bundle_tests PRIVATE /W4 /utf-8 /EHsc)
  else()
    target_link_options(bundle_tests PRIVATE -static)
  endif()
  target_compile_definitions(PortableMixer PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A00000B)
  target_link_libraries(PortableMixer PRIVATE ole32 uuid mmdevapi avrt comctl32 comdlg32 shell32 user32 gdi32 cfgmgr32 advapi32 wintrust)
  if(MSVC)
    target_compile_options(PortableMixer PRIVATE /W4 /permissive- /utf-8 /EHsc)
    target_link_options(PortableMixer PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest")
  else()
    enable_language(RC)
    target_sources(PortableMixer PRIVATE src/app.rc)
    set_source_files_properties(src/app.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest")
    target_include_directories(PortableMixer PRIVATE src)
    target_compile_options(PortableMixer PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Os;-ffunction-sections;-fdata-sections>)
    target_link_options(PortableMixer PRIVATE -static -municode -Wl,--gc-sections,--dynamicbase,--nxcompat,--high-entropy-va,--no-insert-timestamp,--strip-all)
    target_link_options(mixer_core_tests PRIVATE -static)
  endif()
endif()
